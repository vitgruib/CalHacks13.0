const db = await createLocalDb({ world: { seedJson: seed } });

await worker.idle();
const ids: string[] = [];
for (const [i, [, , req]] of cases.entries()) {
  const c = await db.connect();
  await c.call('register_person', { role: 'scientist', name: `Scientist ${i}` });
  await c.call('submit_request', { sharedWith: [], grantId: undefined, ...req });
}
await worker.idle();
const rows = worker.sessions.purchasing.rows('request');
let bad = 0;
for (const [i, [exp, label, req]] of cases.entries()) {
  const r = rows.find((x: any) => x.item === req.item && x.price === req.price);
  const got = ['bought', 'rerouted'].includes(r?.state) ? 'buy' : 'stop';
  const ok = exp === 'either' || exp === got;
  if (!ok) bad++;
  console.log(`${ok ? 'ok ' : 'BAD'} ${label.padEnd(26)} exp=${exp.padEnd(6)} state=${r?.state} on=${r?.chargedGrantId ?? '-'}\n      ${String(r?.explanation).replace(/\s+/g, ' ').slice(0, 260)}`);
}
console.log(`\n${bad} wrong of ${cases.length}`);
